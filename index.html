export async function onRequest(context) { const { request, env } = context; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, PUT, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders }); } // অটো-ডিটেক্ট D1 ডাটাবেজ বাইন্ডিং (retail অথবা DB)[cite: 12] const db = env.retail || env.DB; if (!db) { return new Response(JSON.stringify({ success: false, error: "Database binding not found in environment (env.retail or env.DB)" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } try { const url = new URL(request.url); // ১. GET রিকোয়েস্ট: প্রোডাক্ট বা মোড অনুযায়ী ডাটা ফেচ করা[cite: 12] if (request.method === "GET") { const productCode = url.searchParams.get("code") || url.searchParams.get("id"); const sellerId = url.searchParams.get("seller_id"); const mode = url.searchParams.get("mode"); let query = "SELECT * FROM products"; let params = []; let conditions = []; if (productCode) { conditions.push("(product_code = ? OR id = ?)"); params.push(productCode, productCode); } if (sellerId) { conditions.push("seller_id = ?"); params.push(sellerId); } if (mode) { const lowerMode = mode.toLowerCase().trim(); // রিটেইল মোডের ক্ষেত্রে retail, both, all এবং ফাঁকা/null প্রোডাক্টগুলো একসাথে ফেচ করার সুনির্দিষ্ট লজিক if (lowerMode === 'retail') { conditions.push("(LOWER(upload_mode) = 'retail' OR LOWER(upload_mode) = 'both' OR LOWER(upload_mode) = 'all' OR upload_mode IS NULL OR upload_mode = '')"); } else if (lowerMode === 'dropshipping' || lowerMode === 'dropship') { conditions.push("(LOWER(upload_mode) = 'dropshipping' OR LOWER(upload_mode) = 'dropship' OR LOWER(upload_mode) = 'all' OR LOWER(upload_mode) = 'both' OR upload_mode IS NULL OR upload_mode = '')"); } else if (lowerMode === 'wholesale') { conditions.push("(LOWER(upload_mode) = 'wholesale' OR LOWER(upload_mode) = 'both' OR LOWER(upload_mode) = 'all' OR upload_mode IS NULL OR upload_mode = '')"); } else { conditions.push("LOWER(upload_mode) = ?"); params.push(lowerMode); } } if (conditions.length > 0) { query += " WHERE " + conditions.join(" AND "); } query += " ORDER BY created_at DESC"; const stmt = db.prepare(query).bind(...params); const { results } = await stmt.all(); return new Response(JSON.stringify({ success: true, data: results }), { headers: { ...corsHeaders, "Content-Type": "application/json" } }); } // ২. POST রিকোয়েস্ট: নতুন প্রোডাক্ট ইনপুট দেওয়া[cite: 12] if (request.method === "POST") { const body = await request.json(); const { seller_id, product_code, title, upload_mode, category, price, market_price, dropship_base_price, dropship_resell_price, price_sample, price_min_whole, price_bulk, stock, moq, image_url, images_json, video_url, youtube_video, youtube_live, facebook_live, target_country, discount_rate, allow_custom_note, chat_with_seller, description } = body; const insertQuery = ` INSERT INTO products ( seller_id, product_code, title, upload_mode, category, price, market_price, dropship_base_price, dropship_resell_price, price_sample, price_min_whole, price_bulk, stock, moq, image_url, images_json, video_url, youtube_video, youtube_live, facebook_live, target_country, discount_rate, allow_custom_note, chat_with_seller, description, like_count, created_at ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 0, datetime('now')) `; await db.prepare(insertQuery).bind( seller_id || null, product_code || null, title || null, upload_mode || "retail", category || null, price || 0, market_price || 0, dropship_base_price || 0, dropship_resell_price || 0, price_sample || 0, price_min_whole || 0, price_bulk || 0, stock || 0, moq || 1, image_url || null, images_json ? (typeof images_json === 'string' ? images_json : JSON.stringify(images_json)) : null, video_url || null, youtube_video || null, youtube_live || null, facebook_live || null, target_country || "bd", discount_rate || "0", allow_custom_note ? 1 : 0, chat_with_seller ? 1 : 0, description || null ).run(); return new Response(JSON.stringify({ success: true, message: "Product uploaded successfully!" }), { headers: { ...corsHeaders, "Content-Type": "application/json" } }); } // ৩. PUT রিকোয়েস্ট: লাইক কাউন্ট আপডেট করা[cite: 12] if (request.method === "PUT") { const body = await request.json(); const { product_code } = body; if (!product_code) { return new Response(JSON.stringify({ success: false, error: "Product code is required" }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } const updateQuery = ` UPDATE products SET like_count = COALESCE(like_count, 0) + 1 WHERE product_code = ? OR id = ? `; await db.prepare(updateQuery).bind(product_code, product_code).run(); return new Response(JSON.stringify({ success: true, message: "Like count updated successfully!" }), { headers: { ...corsHeaders, "Content-Type": "application/json" } }); } return new Response(JSON.stringify({ success: false, error: "Method not allowed" }), { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } catch (err) { return new Response(JSON.stringify({ success: false, error: err.message }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } }
