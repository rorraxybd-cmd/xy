// থাম্বনেইল সহ লাইভ প্রিভিউ ফেচ করার ফাংশন async function fetchProductPreview() { let cjProductId = document.getElementById('cjProductId').value.trim(); if (!cjProductId) { alert("প্রথমে প্রোডাক্ট আইডি (PID) দিন।"); return; } const previewBox = document.getElementById('productPreviewBox'); const previewTitle = document.getElementById('previewTitle'); const previewPrice = document.getElementById('previewPrice'); const previewSizes = document.getElementById('previewSizes'); const previewImagesGrid = document.getElementById('previewImagesGrid'); const previewVideoContainer = document.getElementById('previewVideoContainer'); const previewBtn = document.getElementById('previewBtn'); previewBox.style.display = 'block'; previewTitle.innerHTML = 'Loading product details from CJ...'; previewPrice.innerHTML = ''; previewSizes.innerHTML = ''; previewImagesGrid.innerHTML = ''; previewVideoContainer.innerHTML = ''; previewBtn.disabled = true; try { const res = await fetch('/api/import-cj', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ cjProductId, retailMargin: document.getElementById('retailMargin').value, shippingCushion: document.getElementById('shippingCushion').value, wholesaleMargin: document.getElementById('wholesaleMargin').value, previewOnly: true }) }); const result = await res.json(); if (result.success) { previewTitle.innerHTML = result.data.title; // এখানে জিরো দেখানোর সমস্যা সমাধানের জন্য সঠিক কাউন্ট বসানো হয়েছে let imgCount = result.imagesArray ? result.imagesArray.length : (result.data.imagesCount || 0); previewPrice.innerHTML = `Retail Price: $${result.data.retailPrice} | Sample Price: $${result.data.priceSample} | Total Images: ${imgCount}`; if (result.data.sizes && result.data.sizes.length > 0) { previewSizes.innerHTML = `Available Sizes / Variants: ${result.data.sizes.join(', ')}`; } else { previewSizes.innerHTML = `Available Sizes: Standard / Free Size`; } // ছবিগুলোর ছোট ছোট থাম্বনেইল রেন্ডার করা if (result.imagesArray && Array.isArray(result.imagesArray)) { result.imagesArray.forEach(imgUrl => { let img = document.createElement('img'); img.src = imgUrl; img.style.width = '55px'; img.style.height = '55px'; img.style.objectFit = 'cover'; img.style.borderRadius = '6px'; img.style.border = '1px solid rgba(212, 175, 55, 0.3)'; previewImagesGrid.appendChild(img); }); } // ভিডিও থাকলে তা রেন্ডার করা if (result.videoUrl) { previewVideoContainer.innerHTML = `
✓ VIDEO AVAILABLE
`; } else { previewVideoContainer.innerHTML = `
No video available for this product.
`; } } else { previewTitle.innerHTML = `Error: ${result.error}`; } } catch (err) { previewTitle.innerHTML = `Preview Failed: ${err.message}`; } finally { previewBtn.disabled = false; } }
